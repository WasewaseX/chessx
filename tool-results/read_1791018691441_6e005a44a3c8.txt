    26→export function LessonPlayer({ lessonId }: { lessonId: string }) {
    27→  const { navigate, profile } = useApp()
    28→  const found = findLesson(lessonId)
    29→  const [stepIdx, setStepIdx] = useState(0)
    30→  const [canAdvance, setCanAdvance] = useState(false)
    31→  const [done, setDone] = useState(false)
    32→  const [coachOpen, setCoachOpen] = useState(false)
    33→  const savedRef = useRef({ stepsDone: 0, postedDone: false })
    34→
    35→  const level = found?.level
    36→  const lesson = found?.lesson
    37→
    38→  const saveProgress = useCallback(
    39→    (stepsDone: number, finished: boolean, hintNow = false) => {
    40→      if (!lesson || !level) return
    41→      fetch('/api/progress', {
    42→        method: 'POST',
    43→        headers: { 'Content-Type': 'application/json' },
    44→        body: JSON.stringify({
    45→          lessonId: lesson.id,
    46→          stepsDone,
    47→          totalSteps: lesson.steps.length,
    48→          done: finished,
    49→          level: level.n,
    50→          hintNow,
    51→        }),
    52→      })
    53→        .then((r) => r.json())
    54→        .then((d) => {
    55→          if (d.profile) useApp.getState().setProfile(d.profile)
    56→        })
    57→        .catch(() => {})
    58→    },
    59→    [lesson, level],
    60→  )
    61→
    62→  useEffect(() => {
    63→    if (!lesson) return
    64→    if (stepIdx > savedRef.current.stepsDone) {
    65→      savedRef.current.stepsDone = stepIdx
    66→      saveProgress(stepIdx, false)
    67→    }
    68→  }, [stepIdx, lesson, saveProgress])
    69→
    70→  // finishing the last step unlocks completion
    71→  const isLastStep = lesson ? stepIdx === lesson.steps.length - 1 : false
    72→  if (lesson && isLastStep && canAdvance && !done) {
    73→    setDone(true)
    74→    saveProgress(lesson.steps.length, true)
    75→  }
    76→
    77→  if (!lesson || !level) {
    78→    return (
    79→      <div className="mx-auto max-w-2xl px-4 py-16 text-center">
    80→        <p className="text-muted-foreground">Lesson not found.</p>
    81→        <Button className="btn-hero mt-4" onClick={() => navigate('lessons')}>
    82→          Back to lessons
    83→        </Button>
    84→      </div>
    85→    )
    86→  }
    87→
    88→  const step = lesson.steps[stepIdx]
    89→  const last = stepIdx === lesson.steps.length - 1
    90→
    91→  function goNext() {
    92→    if (last && !done) {
    93→      setDone(true)
    94→      saveProgress(lesson.steps.length, true)
    95→      return
    96→    }
    97→    if (last && done) {
    98→      navigate('lessons')
    99→      return
   100→    }
   101→    setStepIdx((i) => i + 1)
   102→    setCanAdvance(false)
   103→  }
   104→
   105→  function goPrev() {
   106→    if (stepIdx > 0) {
   107→      setStepIdx((i) => i - 1)
   108→      setCanAdvance(true)
   109→    }
   110→  }
   111→
   112→  // (completion effect moved above the early return to keep hook order stable)
   113→
   114→  return (
   115→    <div className="mx-auto w-full max-w-6xl px-4 py-4">
   116→      {/* header */}
   117→      <div className="mb-4 flex flex-wrap items-center gap-3">
   118→        <button onClick={() => navigate('lessons')} className="flex items-center gap-1 text-sm font-semibold text-muted-foreground hover:text-foreground">
   119→          <ArrowLeft className="h-4 w-4" /> Lessons
   120→        </button>
   121→        <div className="hidden h-4 w-px bg-border sm:block" />
   122→        <div className="min-w-0">
   123→          <div className="text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   124→            Level {level.n} · {level.title}
   125→          </div>
   126→          <h1 className="truncate font-display text-lg font-bold">{lesson.title}</h1>
   127→        </div>
   128→        <Button variant="secondary" size="sm" className="ml-auto" onClick={() => setCoachOpen(true)}>
   129→          <MessageSquareText className="h-4 w-4" /> Ask the coach
   130→        </Button>
   131→      </div>
   132→
   133→      {/* progress dots */}
   134→      <div className="mb-4 flex items-center gap-1.5">
   135→        {lesson.steps.map((_, i) => (
   136→          <button
   137→            key={i}
   138→            aria-label={`Step ${i + 1}`}
   139→            onClick={() => {
   140→              if (i <= Math.max(stepIdx, savedRef.current.stepsDone)) {
   141→                setStepIdx(i)
   142→                setCanAdvance(i < savedRef.current.stepsDone)
   143→              }
   144→            }}
   145→            className={cn(
   146→              'h-2 rounded-full transition-all',
   147→              i === stepIdx ? 'w-6 bg-primary' : i < stepIdx ? 'w-2 bg-primary/60' : 'w-2 bg-border',
   148→            )}
   149→          />
   150→        ))}
   151→        <span className="ml-2 text-xs font-semibold text-muted-foreground">
   152→          {stepIdx + 1} / {lesson.steps.length}
   153→        </span>
   154→      </div>
   155→
   156→      {step.type === 'playout' ? (
   157→        <PlayoutStepView
   158→          key={stepIdx}
   159→          step={step}
   160→          level={level.n}
   161→          onPass={() => setCanAdvance(true)}
   162→          soundEnabled={profile?.soundEnabled ?? true}
   163→          showLegal={profile?.showLegal ?? true}
   164→          theme={profile?.theme ?? 'green'}
   165→        />
   166→      ) : step.type === 'exercise' ? (
   167→        <ExerciseView
   168→          key={stepIdx}
   169→          step={step}
   170→          lessonTitle={lesson.title}
   171→          onPass={() => setCanAdvance(true)}
   172→          soundEnabled={profile?.soundEnabled ?? true}
   173→          showLegal={profile?.showLegal ?? true}
   174→          theme={profile?.theme ?? 'green'}
   175→          onAskCoach={() => setCoachOpen(true)}
   176→        />
   177→      ) : (
   178→        <div className="grid gap-5 lg:grid-cols-[1fr_420px]">
   179→          <div className={cn(step.type === 'demo' ? 'order-2 lg:order-1' : 'hidden')}>
   180→            {step.type === 'demo' && (
   181→              <DemoBoard
   182→                key={stepIdx}
   183→                fen={step.fen}
   184→                moves={step.moves}
   185→                marks={step.marks}
   186→                arrows={step.arrows}
   187→                caption={step.caption}
   188→                soundEnabled={profile?.soundEnabled ?? true}
   189→              />
   190→            )}
   191→          </div>
   192→          <div className={cn('order-1 lg:order-2', step.type !== 'demo' && 'lg:col-span-2')}>
   193→            {step.type === 'text' && <TextStepView step={step} onReady={() => setCanAdvance(true)} />}
   194→            {step.type === 'demo' && <DemoTextView step={step} onReady={() => setCanAdvance(true)} />}
   195→            {step.type === 'quiz' && <QuizStepView key={stepIdx} step={step} onPass={() => setCanAdvance(true)} />}
   196→          </div>
   197→        </div>
   198→      )}
   199→
   200→      {/* footer nav */}
   201→      <div className="mt-6 flex items-center justify-between">
   202→        <Button variant="secondary" onClick={goPrev} disabled={stepIdx === 0}>
   203→          <ChevronLeft className="h-4 w-4" /> Back
   204→        </Button>
   205→        {canAdvance ? (
   206→          <Button className="btn-hero px-8" onClick={goNext}>
   207→            {last && done ? 'Finish' : last ? 'Complete lesson' : 'Continue'} <ChevronRight className="h-4 w-4" />
   208→          </Button>
   209→        ) : (
   210→          <Button variant="secondary" disabled>
   211→            {step.type === 'exercise' || step.type === 'playout' ? 'Solve it to continue' : step.type === 'quiz' ? 'Answer to continue' : '…'}
   212→          </Button>
   213→        )}
   214→      </div>
   215→
   216→      <CoachDrawer
   217→        open={coachOpen}
   218→        onOpenChange={setCoachOpen}
   219→        context={{
   220→          lessonTitle: lesson.title,
   221→          fen: step.type === 'demo' || step.type === 'exercise' || step.type === 'playout' ? step.fen : undefined,
   222→          stepHint: step.type === 'exercise' ? `${step.goal}. Do not reveal the solution move directly.` : undefined,
   223→          skillLevel: profile?.skillLevel ?? 'beginner',
   224→        }}
   225→      />
   226→    </div>
   227→  )
   228→}

   273→function DemoBoard({
   274→  fen,
   275→  moves,
   276→  marks,
   277→  arrows,
   278→  caption,
   279→  soundEnabled,
   280→}: {
   281→  fen: string
   282→  moves?: string[]
   283→  marks?: Mark[]
   284→  arrows?: Arrow[]
   285→  caption?: string
   286→  soundEnabled: boolean
   287→}) {
   288→  const [shownFen, setShownFen] = useState(fen)
   289→  const [ply, setPly] = useState(0)
   290→  const [lastMove, setLastMove] = useState<{ from: string; to: string } | null>(null)
   291→  const gameRef = useRef(new Chess(fen))
   292→
   293→  useEffect(() => {
   294→    gameRef.current = new Chess(fen)
   295→    setShownFen(fen)
   296→    setPly(0)
   297→    setLastMove(null)
   298→  }, [fen])
   299→
   300→  const stepForward = useCallback(() => {
   301→    if (!moves || ply >= moves.length) return
   302→    const g = gameRef.current
   303→    const mv = g.move(moves[ply])
   304→    if (mv) {
   305→      setShownFen(g.fen())
   306→      setPly((p) => p + 1)
   307→      setLastMove({ from: mv.from, to: mv.to })
   308→      playSound(mv.captured ? 'capture' : 'move', soundEnabled)
   309→    }
   310→  }, [moves, ply, soundEnabled])
   311→
   312→  // auto-play the line when the step appears
   313→  useEffect(() => {
   314→    if (!moves?.length) return
   315→    let p = 0
   316→    const timers: ReturnType<typeof setTimeout>[] = []
   317→    const tick = () => {
   318→      stepForwardRef.current()
   319→      p++
   320→      if (p < moves.length) timers.push(setTimeout(tick, 800))
   321→    }
   322→    timers.push(setTimeout(tick, 700))
   323→    return () => timers.forEach(clearTimeout)
   324→     
   325→  }, [fen])
   326→
   327→  const stepForwardRef = useRef(stepForward)
   328→  stepForwardRef.current = stepForward
   329→
   330→  const checkSquare = useMemo(() => {
   331→    const g = new Chess(shownFen)
   332→    if (!g.isCheck()) return null
   333→    const turn = g.turn()
   334→    return g.board().flat().find((s) => s && s.type === 'k' && s.color === turn)?.square ?? null
   335→  }, [shownFen])
   336→
   337→  return (
   338→    <div>
   339→      <ChessBoard fen={shownFen} lastMove={lastMove} marks={marks} arrows={arrows} checkSquare={checkSquare} interactive={false} />
   340→      <div className="mt-2 flex items-center justify-between">
   341→        <div className="text-sm text-muted-foreground">{caption ?? ''}</div>
   342→        {moves && moves.length > 0 && (
   343→          <Button
   344→            variant="secondary"
   345→            size="sm"
   346→            onClick={() => {
   347→              gameRef.current = new Chess(fen)
   348→              setShownFen(fen)
   349→              setPly(0)
   350→              setLastMove(null)
   351→              setTimeout(() => {
   352→                let p = 0
   353→                const tick = () => {
   354→                  stepForwardRef.current()
   355→                  p++
   356→                  if (p < (moves?.length ?? 0)) setTimeout(tick, 800)
   357→                }
   358→                setTimeout(tick, 400)
   359→              }, 50)
   360→            }}
   361→          >
   362→            <RotateCcw className="h-4 w-4" /> Replay
   363→          </Button>
   364→        )}
   365→      </div>
   366→    </div>
   367→  )
   368→}

   370→function QuizStepView({ step, onPass }: { step: Extract<LessonStep, { type: 'quiz' }>; onPass: () => void }) {
   371→  const [chosen, setChosen] = useState<number | null>(null)
   372→  const [answeredCorrect, setAnsweredCorrect] = useState(false)
   373→  const correctIdx = step.options.findIndex((o) => o.correct)
   374→
   375→  function choose(i: number) {
   376→    if (answeredCorrect) return
   377→    setChosen(i)
   378→    if (step.options[i].correct) {
   379→      setAnsweredCorrect(true)
   380→      onPass()
   381→    }
   382→  }
   383→
   384→  return (
   385→    <div className="rounded-lg bg-card p-6 shadow-sm">
   386→      <h2 className="font-display text-xl font-bold">{step.title}</h2>
   387→      {step.body && <p className="mt-2 text-sm text-muted-foreground">{step.body}</p>}
   388→      <p className="mt-3 font-semibold">{step.question}</p>
   389→      <div className="mt-4 grid gap-2">
   390→        {step.options.map((o, i) => {
   391→          const isChosen = chosen === i
   392→          const state = !isChosen ? 'idle' : o.correct ? 'correct' : 'wrong'
   393→          return (
   394→            <button
   395→              key={i}
   396→              onClick={() => choose(i)}
   397→              disabled={answeredCorrect}
   398→              className={cn(
   399→                'flex items-start gap-3 rounded-md border px-4 py-3 text-left text-sm font-medium transition',
   400→                state === 'idle' && 'border-border hover:border-primary/50 hover:bg-accent/50',
   401→                state === 'correct' && 'border-primary bg-primary/10',
   402→                state === 'wrong' && 'border-destructive bg-destructive/10',
   403→                answeredCorrect && !isChosen && 'opacity-50',
   404→              )}
   405→            >
   406→              {state === 'correct' ? (
   407→                <CheckCircle2 className="mt-0.5 h-4 w-4 shrink-0 text-primary" />
   408→              ) : state === 'wrong' ? (
   409→                <XCircle className="mt-0.5 h-4 w-4 shrink-0 text-destructive" />
   410→              ) : (
   411→                <span className="mt-0.5 h-4 w-4 shrink-0 rounded-full border-2 border-muted-foreground/40" />
   412→              )}
   413→              <span>
   414→                {o.text}
   415→                {isChosen && <span className="mt-1 block text-xs font-normal text-muted-foreground">{o.why}</span>}
   416→              </span>
   417→            </button>
   418→          )
   419→        })}
   420→      </div>
   421→      {answeredCorrect && (
   422→        <p className="mt-3 text-sm font-semibold text-primary">Correct. {step.options[correctIdx]?.why}</p>
   423→      )}
   424→    </div>
   425→  )
   426→}

   428→function ExerciseView({
   429→  step,
   430→  lessonTitle,
   431→  onPass,
   432→  soundEnabled,
   433→  showLegal,
   434→  theme,
   435→  onAskCoach,
   436→}: {
   437→  step: ExerciseStep
   438→  lessonTitle: string
   439→  onPass: () => void
   440→  soundEnabled: boolean
   441→  showLegal: boolean
   442→  theme: string
   443→  onAskCoach: () => void
   444→}) {
   445→  const gameRef = useRef(new Chess(step.fen))
   446→  const [fen, setFen] = useState(step.fen)
   447→  const [movesSoFar, setMovesSoFar] = useState<string[]>([])
   448→  const [status, setStatus] = useState<'solving' | 'wrong' | 'done'>('solving')
   449→  const [hintShown, setHintShown] = useState(false)
   450→  const [shake, setShake] = useState(false)
   451→  const [lastMove, setLastMove] = useState<{ from: string; to: string } | null>(null)
   452→
   453→  const expectedUserIdx = movesSoFar.length // next user move is solution[movesSoFar.length] (even index)
   454→  const game = useMemo(() => new Chess(fen), [fen])
   455→  const sideToMove = game.turn()
   456→  const checkSquare = useMemo(() => {
   457→    if (!game.isCheck()) return null
   458→    return game.board().flat().find((s) => s && s.type === 'k' && s.color === game.turn())?.square ?? null
   459→  }, [game])
   460→
   461→  function reset() {
   462→    gameRef.current = new Chess(step.fen)
   463→    setFen(step.fen)
   464→    setMovesSoFar([])
   465→    setStatus('solving')
   466→    setLastMove(null)
   467→  }
   468→
   469→  function onMove(from: Square, to: Square, promotion?: string) {
   470→    if (status === 'done') return
   471→    const g = gameRef.current
   472→    const expected = step.solution[movesSoFar.length]
   473→    let mv
   474→    try {
   475→      mv = g.move({ from, to, promotion: promotion ?? undefined })
   476→    } catch {
   477→      return
   478→    }
   479→    if (!mv) return
   480→
   481→    // is it the expected move?
   482→    let correct = false
   483→    if (expected) {
   484→      const expectedNorm = expected.replace(/[+#]/g, '')
   485→      const mvNorm = mv.san.replace(/[+#]/g, '')
   486→      correct = mvNorm === expectedNorm
   487→    }
   488→
   489→    if (!correct) {
   490→      g.undo()
   491→      setStatus('wrong')
   492→      setShake(true)
   493→      playSound('wrong', soundEnabled)
   494→      setTimeout(() => setShake(false), 450)
   495→      return
   496→    }
   497→
   498→    setMovesSoFar((m) => [...m, mv.san])
   499→    setLastMove({ from: mv.from, to: mv.to })
   500→    setFen(g.fen())
   501→    playSound(mv.captured ? 'capture' : 'move', soundEnabled)
   502→
   503→    // opponent scripted reply
   504→    const reply = step.solution[movesSoFar.length + 1]
   505→    if (reply) {
   506→      setTimeout(() => {
   507→        try {
   508→          const rmv = g.move(reply)
   509→          if (rmv) {
   510→            setMovesSoFar((m) => [...m, rmv.san])
   511→            setLastMove({ from: rmv.from, to: rmv.to })
   512→            setFen(g.fen())
   513→            playSound(rmv.captured ? 'capture' : 'move', soundEnabled)
   514→          }
   515→        } catch {
   516→          /* scripted reply must be legal; validator guarantees */
   517→        }
   518→      }, 550)
   519→    } else {
   520→      // line finished
   521→      setStatus('done')
   522→      playSound('correct', soundEnabled)
   523→      onPass()
   524→    }
   525→    // if reply exists but it was the last move of the line
   526→    if (reply && movesSoFar.length + 2 >= step.solution.length) {
   527→      setTimeout(() => {
   528→        setStatus('done')
   529→        playSound('correct', soundEnabled)
   530→        onPass()
   531→      }, 1100)
   532→    }
   533→  }
   534→
   535→  const userSideFromFen = useMemo(() => new Chess(step.fen).turn(), [step.fen])
   536→
   537→  return (
   538→    <div className="grid gap-5 lg:grid-cols-[1fr_400px]">
   539→      <div>
   540→        <ChessBoard
   541→          fen={fen}
   542→          orientation={userSideFromFen}
   543→          onMove={onMove}
   544→          movableSide={status === 'done' ? undefined : userSideFromFen}
   545→          interactive={status !== 'done'}
   546→          lastMove={lastMove}
   547→          checkSquare={checkSquare}
   548→          showLegal={showLegal && status !== 'done'}
   549→          theme={theme}
   550→          shake={shake}
   551→        />
   552→      </div>
   553→      <div className="flex flex-col gap-3">
   554→        <div className="rounded-lg bg-card p-5 shadow-sm">
   555→          <div className="text-xs font-semibold uppercase tracking-wide text-muted-foreground">Exercise</div>
   556→          <h2 className="mt-1 font-display text-xl font-bold">{step.title}</h2>
   557→          {step.body?.map((p, i) => (
   558→            <p key={i} className="mt-2 text-sm text-foreground/90">
   559→              {p}
   560→            </p>
   561→          ))}
   562→          <div className="mt-3 rounded-md bg-secondary px-3 py-2 text-sm font-semibold">{step.goal}</div>
   563→
   564→          {status === 'solving' && (
   565→            <div className="mt-3 text-sm text-muted-foreground">
   566→              {movesSoFar.length > 0 ? (
   567→                <>Line so far: <span className="font-mono font-semibold text-foreground">{movesSoFar.join(' ')}</span></>
   568→              ) : (
   569→                'Your move.'
   570→              )}
   571→            </div>
   572→          )}
   573→          {status === 'wrong' && (
   574→            <div className="mt-3 rounded-md border border-destructive/40 bg-destructive/10 px-3 py-2 text-sm font-semibold text-destructive">
   575→              Not that. Look again — what does the position really want?
   576→            </div>
   577→          )}
   578→          {status === 'done' && (
   579→            <div className="mt-3 rounded-md border border-primary/40 bg-primary/10 px-3 py-2 text-sm font-semibold text-foreground">
   580→              <Trophy className="mr-1 inline h-4 w-4 text-primary" />
   581→              {step.success}
   582→            </div>
   583→          )}
   584→          {step.explanation && status === 'done' && (
   585→            <p className="mt-2 text-sm text-muted-foreground">{step.explanation}</p>
   586→          )}
   587→
   588→          <div className="mt-4 flex gap-2">
   589→            <Button variant="secondary" size="sm" onClick={reset} disabled={status === 'done'}>
   590→              <RotateCcw className="h-4 w-4" /> Reset
   591→            </Button>
   592→            <Button
   593→              variant="secondary"
   594→              size="sm"
   595→              onClick={() => setHintShown(true)}
   596→              disabled={hintShown || status === 'done'}
   597→            >
   598→              <Lightbulb className="h-4 w-4" /> Hint
   599→            </Button>
   600→            <Button variant="secondary" size="sm" onClick={onAskCoach}>
   601→              <MessageSquareText className="h-4 w-4" /> Coach
   602→            </Button>
   603→          </div>
   604→          {hintShown && status !== 'done' && (
   605→            <div className="mt-3 rounded-md border-l-4 border-[#e6a82c] bg-[#e6a82c]/10 px-3 py-2 text-sm">
   606→              {step.hint}
   607→            </div>
   608→          )}
   609→        </div>
   610→      </div>
   611→    </div>
   612→  )

   615→function PlayoutStepView({
   616→  step,
   617→  level,
   618→  onPass,
   619→  soundEnabled,
   620→  showLegal,
   621→  theme,
   622→}: {
   623→  step: Extract<LessonStep, { type: 'playout' }>
   624→  level: number
   625→  onPass: () => void
   626→  soundEnabled: boolean
   627→  showLegal: boolean
   628→  theme: string
   629→}) {
   630→  const gameRef = useRef(new Chess(step.fen))
   631→  const [fen, setFen] = useState(step.fen)
   632→  const [moves, setMoves] = useState<string[]>([])
   633→  const [lastMove, setLastMove] = useState<{ from: string; to: string } | null>(null)
   634→  const [status, setStatus] = useState<'playing' | 'won' | 'lost' | 'draw'>('playing')
   635→  const [thinking, setThinking] = useState(false)
   636→  const judgedRef = useRef(false)
   637→
   638→  const game = useMemo(() => new Chess(fen), [fen])
   639→  const checkSquare = useMemo(() => {
   640→    if (!game.isCheck()) return null
   641→    return game.board().flat().find((s) => s && s.type === 'k' && s.color === game.turn())?.square ?? null
   642→  }, [game])
   643→
   644→  const materialBalance = (g: Chess) => {
   645→    const vals: Record<string, number> = { p: 1, n: 3, b: 3, r: 5, q: 9 }
   646→    let bal = 0
   647→    for (const row of g.board()) {
   648→      for (const sq of row) {
   649→        if (!sq || sq.type === 'k') continue
   650→        bal += sq.color === step.side ? vals[sq.type] : -vals[sq.type]
   651→      }
   652→    }
   653→    return bal
   654→  }
   655→
   656→  const castled = (g: Chess) => {
   657→    // castle rights gone + king on g1/c1 (or g8/c8) counts as castled
   658→    const kingSquare = g.board().flat().find((s) => s && s.type === 'k' && s.color === step.side)?.square
   659→    if (!kingSquare) return false
   660→    const homeRow = step.side === 'w' ? '1' : '8'
   661→    const lostRights = step.side === 'w' ? !g.get('e1' as Square) || true : true
   662→    void lostRights
   663→    return (kingSquare === `g${homeRow}` || kingSquare === `c${homeRow}`) && moves.some((m) => m.startsWith('O-O'))
   664→  }
   665→
   666→  const judge = useCallback(
   667→    (g: Chess): 'won' | 'lost' | 'draw' | null => {
   668→      if (g.isCheckmate()) {
   669→        const matedSide = g.turn()
   670→        return matedSide === step.side ? 'lost' : 'won'
   671→      }
   672→      if (g.isStalemate() || g.isInsufficientMaterial() || g.isThreefoldRepetition() || g.isDraw()) {
   673→        return step.success === 'draw' ? 'won' : 'draw'
   674→      }
   675→      const plyLimit = (step.maxMoves ?? 12) * 2
   676→      if (moves.length >= plyLimit) {
   677→        if (step.success === 'castle') return castled(g) ? 'won' : 'lost'
   678→        if (step.success === 'material') return materialBalance(g) >= 3 ? 'won' : 'lost'
   679→        return 'draw'
   680→      }
   681→      // early material check
   682→      if (step.success !== 'draw') {
   683→        const bal = materialBalance(g)
   684→        if (bal <= -3) return 'lost'
   685→      }
   686→      if (step.success === 'castle' && castled(g)) return 'won'
   687→      if (step.success === 'material' && materialBalance(g) >= 3 && moves.length >= 2) return 'won'
   688→      return null
   689→    },
   690→     
   691→    [fen, moves],
   692→  )
   693→
   694→  useEffect(() => {
   695→    if (status !== 'playing' || judgedRef.current) return
   696→    const result = judge(gameRef.current)
   697→    if (result) {
   698→      judgedRef.current = true
   699→      setStatus(result)
   700→      if (result === 'won') {
   701→        playSound('correct', soundEnabled)
   702→        onPass()
   703→      } else {
   704→        playSound('wrong', soundEnabled)
   705→      }
   706→    }
   707→  }, [fen, status, judge, onPass, soundEnabled])
   708→
   709→  const engineMove = useCallback(async () => {
   710→    const g = gameRef.current
   711→    if (g.isGameOver()) return
   712→    setThinking(true)
   713→    try {
   714→      const { uci } = await engine.bestMove({ fen: g.fen(), skill: Math.max(0, step.engineLevel * 4 - 2), depth: Math.max(2, step.engineLevel + 1), minTime: 350 })
   715→      if (uci && uci.length >= 4) {
   716→        const mv = g.move({ from: uci.slice(0, 2), to: uci.slice(2, 4), promotion: uci.slice(4, 5) || undefined })
   717→        if (mv) {
   718→          setFen(g.fen())
   719→          setMoves((m) => [...m, mv.san])
   720→          setLastMove({ from: mv.from, to: mv.to })
   721→          playSound(mv.captured ? 'capture' : 'move', soundEnabled)
   722→        }
   723→      }
   724→    } finally {
   725→      setThinking(false)
   726→    }
   727→  }, [step.engineLevel, soundEnabled])
   728→
   729→  function onMove(from: Square, to: Square, promotion?: string) {
   730→    if (status !== 'playing') return
   731→    const g = gameRef.current
   732→    try {
   733→      const mv = g.move({ from, to, promotion: promotion ?? undefined })
   734→      if (!mv) return
   735→      setFen(g.fen())
   736→      setMoves((m) => [...m, mv.san])
   737→      setLastMove({ from: mv.from, to: mv.to })
   738→      playSound(mv.captured ? 'capture' : 'move', soundEnabled)
   739→      if (!g.isGameOver()) void engineMove()
   740→    } catch {
   741→      /* illegal */
   742→    }
   743→  }
   744→
   745→  function reset() {
   746→    gameRef.current = new Chess(step.fen)
   747→    setFen(step.fen)
   748→    setMoves([])
   749→    setLastMove(null)
   750→    setStatus('playing')
   751→    judgedRef.current = false
   752→  }
   753→
   754→  return (
   755→    <div className="grid gap-5 lg:grid-cols-[1fr_400px]">
   756→      <div>
   757→        <ChessBoard
   758→          fen={fen}
   759→          orientation={step.side}
   760→          onMove={onMove}
   761→          movableSide={status === 'playing' ? step.side : undefined}
   762→          interactive={status === 'playing'}
   763→          lastMove={lastMove}
   764→          checkSquare={checkSquare}
   765→          showLegal={showLegal && status === 'playing'}
   766→          theme={theme}
   767→        />
   768→      </div>
   769→      <div className="rounded-lg bg-card p-5 shadow-sm">
   770→        <div className="text-xs font-semibold uppercase tracking-wide text-muted-foreground">Play it out · Level {level}</div>
   771→        <h2 className="mt-1 font-display text-xl font-bold">{step.title}</h2>
   772→        {step.body?.map((p, i) => (
   773→          <p key={i} className="mt-2 text-sm text-foreground/90">
   774→            {p}
   775→          </p>
   776→        ))}
   777→        <div className="mt-3 rounded-md bg-secondary px-3 py-2 text-sm font-semibold">{step.goal}</div>
   778→        <div className="mt-2 text-xs text-muted-foreground">
   779→          {thinking ? 'Opponent thinking…' : moves.length > 0 ? `Moves played: ${moves.length}` : 'Your move.'}
   780→        </div>
   781→
   782→        {status === 'won' && (
   783→          <div className="mt-3 rounded-md border border-primary/40 bg-primary/10 px-3 py-2 text-sm font-semibold">
   784→            <Trophy className="mr-1 inline h-4 w-4 text-primary" />
   785→            {step.successText}
   786→          </div>
   787→        )}
   788→        {(status === 'lost' || status === 'draw') && (
   789→          <div className="mt-3 rounded-md border border-destructive/40 bg-destructive/10 px-3 py-2 text-sm font-semibold text-destructive">
   790→            {step.failText ?? (status === 'draw' ? 'Draw — not the goal here.' : 'That did not work. Reset and try a different plan.')}
   791→          </div>
   792→        )}
   793→
   794→        <div className="mt-4 flex gap-2">
   795→          <Button variant="secondary" size="sm" onClick={reset} disabled={status === 'playing' && moves.length === 0}>
   796→            <RotateCcw className="h-4 w-4" /> Reset
   797→          </Button>
   798→        </div>
   799→      </div>
   800→    </div>
   801→  )
   802→}
